cmake_minimum_required(VERSION 3.24)
project(scriber_localvqe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
include(FetchContent)
# Network access is build-time only. Every input is immutable and hash checked.
FetchContent_Declare(localvqe URL "${SOURCE_URL}" URL_HASH "SHA256=${SOURCE_SHA256}" TLS_VERIFY ON)
FetchContent_Declare(ggml_src URL "${GGML_URL}" URL_HASH "SHA256=${GGML_SHA256}" TLS_VERIFY ON SOURCE_SUBDIR unused)
FetchContent_MakeAvailable(localvqe ggml_src)
set(MODEL "${CMAKE_CURRENT_BINARY_DIR}/model.gguf")
if(EXISTS "${MODEL}")
    file(SHA256 "${MODEL}" model_hash)
endif()
if(NOT model_hash STREQUAL MODEL_SHA256)
    file(DOWNLOAD "${MODEL_URL}" "${MODEL}" EXPECTED_HASH "SHA256=${MODEL_SHA256}" TLS_VERIFY ON STATUS download_status)
endif()

# One statically linked CPU backend; no DLL discovery, GPU, OpenMP runtime,
# native-machine tuning. x64 inference uses AVX2/FMA/F16C, checked by Rust
# before entering native code; this is a fixed portable target, never -march=native.
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
foreach(isa SSE42 AVX AVX2 FMA F16C AVX512 BMI2)
    set(GGML_${isa} OFF CACHE BOOL "" FORCE)
endforeach()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "AMD64|amd64|x86_64")
    foreach(isa AVX AVX2 FMA F16C)
        set(GGML_${isa} ON CACHE BOOL "" FORCE)
    endforeach()
endif()
add_subdirectory("${ggml_src_SOURCE_DIR}" ggml-build EXCLUDE_FROM_ALL)

# The upstream all-model build enables dynamic backends and optional models.
# Compile only the v1.x graph and its dependencies with a narrow static bridge.
set(vqe "${localvqe_SOURCE_DIR}/ggml")
file(READ "${vqe}/localvqe_graph.cpp" graph)
string(FIND "${graph}" "void ensure_backends_loaded() {" start)
string(FIND "${graph}" "void dvqe_list_devices(" end)
if(start LESS 0 OR end LESS 0)
    message(FATAL_ERROR "LocalVQE backend patch no longer matches")
endif()
string(SUBSTRING "${graph}" 0 ${start} prefix)
string(SUBSTRING "${graph}" ${end} -1 suffix)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/localvqe_graph.cpp" "${prefix}void ensure_backends_loaded() {}\n\n${suffix}")
# ggml_fopen handles UTF-8 Windows paths, including non-ASCII user/temp names.
foreach(unit localvqe_graph model_hash)
    if(unit STREQUAL "localvqe_graph")
        set(input "${CMAKE_CURRENT_BINARY_DIR}/${unit}.cpp")
    else()
        set(input "${vqe}/${unit}.cpp")
    endif()
    file(READ "${input}" source)
    string(REPLACE "std::fopen(" "ggml_fopen(" source "${source}")
    string(REPLACE "= fopen(" "= ggml_fopen(" source "${source}")
    if(unit STREQUAL "model_hash")
        string(REPLACE "if (bypass[0] == '1') return true;" "(void)bypass; // Scriber never bypasses integrity verification." source "${source}")
    endif()
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${unit}.cpp" "#include \"ggml.h\"\n${source}")
endforeach()
file(READ "${vqe}/localvqe_api.cpp" api)
# stdout is the sidecar JSON protocol. Model construction must remain quiet.
string(REPLACE "backend_name, device_index, true, n_threads" "backend_name, device_index, false, n_threads" api "${api}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/localvqe_api.cpp" "${api}")
add_library(scriber_localvqe STATIC bridge.cpp
    "${CMAKE_CURRENT_BINARY_DIR}/localvqe_api.cpp"
    "${CMAKE_CURRENT_BINARY_DIR}/localvqe_graph.cpp"
    "${CMAKE_CURRENT_BINARY_DIR}/model_hash.cpp"
    "${vqe}/common.cpp" "${vqe}/daf_frontend.cpp")
target_include_directories(scriber_localvqe PRIVATE "${vqe}")
target_compile_definitions(scriber_localvqe PRIVATE LOCALVQE_BUILD NOMINMAX)
target_link_libraries(scriber_localvqe PRIVATE ggml)
if(MSVC)
    target_compile_options(scriber_localvqe PRIVATE /utf-8 /fp:precise)
else()
    target_compile_options(scriber_localvqe PRIVATE -ffp-contract=off)
endif()
install(TARGETS scriber_localvqe ggml ggml-cpu ggml-base ARCHIVE DESTINATION lib)
install(FILES "${MODEL}" DESTINATION .)
install(FILES "${vqe}/tests/fixtures/regression_input.f32"
    "${vqe}/tests/fixtures/localvqe-v1.3-4.8M-f32.out.f32" DESTINATION fixtures)
